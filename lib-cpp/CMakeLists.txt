cmake_minimum_required(VERSION 3.20)
project(MayhemCore VERSION 0.1.0 LANGUAGES CXX)
add_library(mayhem_core INTERFACE)
target_include_directories(mayhem_core INTERFACE include)
target_compile_features(mayhem_core INTERFACE cxx_std_17)
include(CTest)
if(BUILD_TESTING)
  add_executable(mayhem_controller_test tests/controller_test.cpp)
  target_link_libraries(mayhem_controller_test PRIVATE mayhem_core)
  add_test(NAME pose_controller COMMAND mayhem_controller_test)
endif()
if(BUILD_TESTING)
  add_executable(mayhem_runner_test tests/runner_test.cpp)
  target_link_libraries(mayhem_runner_test PRIVATE mayhem_core)
  add_test(NAME auto_runner COMMAND mayhem_runner_test)
endif()
