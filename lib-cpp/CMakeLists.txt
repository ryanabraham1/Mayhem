cmake_minimum_required(VERSION 3.20)
project(MayhemCore VERSION 0.1.0 LANGUAGES CXX)
add_library(mayhem_core INTERFACE)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
add_library(Mayhem::Core ALIAS mayhem_core)
set_target_properties(mayhem_core PROPERTIES EXPORT_NAME Core)
target_include_directories(mayhem_core INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_features(mayhem_core INTERFACE cxx_std_17)
include(CTest)
if(BUILD_TESTING)
  foreach(name controller runner alliance state_machine)
    add_executable(mayhem_${name}_test tests/${name}_test.cpp)
    target_link_libraries(mayhem_${name}_test PRIVATE mayhem_core)
    add_test(NAME ${name} COMMAND mayhem_${name}_test)
  endforeach()
endif()

install(TARGETS mayhem_core EXPORT MayhemCoreTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT MayhemCoreTargets FILE MayhemCoreTargets.cmake NAMESPACE Mayhem:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MayhemCore)
configure_package_config_file(cmake/MayhemCoreConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/MayhemCoreConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MayhemCore)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/MayhemCoreConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/MayhemCoreConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/MayhemCoreConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MayhemCore)
